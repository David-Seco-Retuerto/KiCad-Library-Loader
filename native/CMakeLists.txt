cmake_minimum_required(VERSION 3.16)
project(KiCadLoaderStepMesher VERSION 0.1.0 LANGUAGES CXX)
find_package(OpenCASCADE 7.9.3 REQUIRED)
add_executable(kicad-step-mesher step_mesher.cpp)
target_compile_features(kicad-step-mesher PRIVATE cxx_std_17)
target_include_directories(kicad-step-mesher PRIVATE ${OpenCASCADE_INCLUDE_DIR})
target_link_directories(kicad-step-mesher PRIVATE ${OpenCASCADE_LIBRARY_DIR})
target_link_libraries(kicad-step-mesher PRIVATE TKDESTEP TKXSBase TKMesh TKPrim TKBRep TKTopAlgo TKG3d TKMath TKernel)
if(WIN32)
  target_compile_definitions(kicad-step-mesher PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(UNIX AND NOT APPLE)
  # Inherit the helper-relative library path for OCCT dependency chains.
  target_link_options(kicad-step-mesher PRIVATE "LINKER:--disable-new-dtags")
  set_target_properties(kicad-step-mesher PROPERTIES INSTALL_RPATH "$ORIGIN/../lib" BUILD_WITH_INSTALL_RPATH FALSE)
endif()
install(TARGETS kicad-step-mesher RUNTIME DESTINATION bin)
